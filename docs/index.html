import jax
import jax.numpy as jp

SCALES = {
    "hand_reach": 0.0,      # disable base reach term (conflicts with raising hand above head)
    "hand_height": 2.0,     # reward right hand above head (z in pelvis frame)
    "hand_wave_y": 1.5,     # reward side-to-side waving (y in pelvis frame)
}

def extra_rewards(env, data, action, info):
    # Pelvis frame
    pelvis = data.site_xpos[env._pelvis_imu_site_id]
    R = data.site_xmat[env._pelvis_imu_site_id].reshape(3, 3)
    # Right hand in pelvis frame (x forward, y left, z up)
    rh_world = data.site_xpos[env._hands_site_id[1]]
    rh_pelvis = R.T @ (rh_world - pelvis)
    hand_z = rh_pelvis[2]
    hand_y = rh_pelvis[1]

    # 1) Height tracking: target above head (~0.648), below max (0.792)
    height_target = 0.70
    height_err = hand_z - height_target
    height_reward = jp.exp(-height_err**2 / (0.6**2))  # wide sigma: signal from rest (z≈0.02)

    # 2) Waving: sinusoidal y target at ~1.5 Hz, amplitude 0.15m around y≈-0.2
    freq = 1.5
    y_center = -0.20
    amp = 0.15
    y_target = y_center + amp * jp.sin(2.0 * jp.pi * freq * data.time)
    wave_err = hand_y - y_target
    wave_reward = jp.exp(-wave_err**2 / (0.12**2))  # sigma allows ~0.15m error still >0.5

    return {
        "hand_height": height_reward,
        "hand_wave_y": wave_reward,
    }

import jax
import jax.numpy as jp

SCALES = {
    "hand_reach": 0.0,       # disable base reach term
    "hand_height": 2.0,      # precision tracking near target (above head)
    "hand_rise": 1.0,        # dense shaping: reward any upward progress from rest
    "hand_wave_y": 2.0,      # side-to-side waving (tighter tracking)
}

def extra_rewards(env, data, action, info):
    pelvis = data.site_xpos[env._pelvis_imu_site_id]
    R = data.site_xmat[env._pelvis_imu_site_id].reshape(3, 3)
    rh_world = data.site_xpos[env._hands_site_id[1]]
    rh_pelvis = R.T @ (rh_world - pelvis)
    hand_z = rh_pelvis[2]
    hand_y = rh_pelvis[1]

    # 1) Precision height tracking: target above head (0.70m), tight sigma for final accuracy
    height_target = 0.70
    height_err = hand_z - height_target
    height_reward = jp.exp(-height_err**2 / (0.15**2))  # sigma=0.15m: strong gradient near target

    # 2) Dense shaping term: grows monotonically from rest (z≈0.02) to target
    # tanh((z - z_rest)/scale) ~0 at rest, ~0.73 at 0.3m, ~0.98 at 0.7m
    z_rest = 0.022
    rise_reward = jp.tanh((hand_z - z_rest) / 0.3)

    # 3) Waving: sinusoidal y target at 1.5 Hz, amplitude 0.15m around y≈-0.20
    freq = 1.5
    y_center = -0.20
    amp = 0.15
    y_target = y_center + amp * jp.sin(2.0 * jp.pi * freq * data.time)
    wave_err = hand_y - y_target
    wave_reward = jp.exp(-wave_err**2 / (0.10**2))  # tighter sigma=0.10m to encourage full amplitude

    return {
        "hand_height": height_reward,
        "hand_rise": rise_reward,
        "hand_wave_y": wave_reward,
    }

    # 4) Waving target: 1.0 Hz, center at shoulder line (-0.1), amplitude 0.25
    freq = 1.0
    y_center = -0.1
    amp = 0.25
    y_target = y_center + amp * jp.sin(2.0 * jp.pi * freq * data.time)
    wave_err = hand_y - y_target
    wave_reward = jp.exp(-wave_err**2 / (0.12**2))
    wave_reward_gated = wave_reward * height_gate

    # 5) Above-head bonus: sigmoid at 0.62m (lowered from 0.65) to trigger before head top
    above_x = 20.0 * (hand_z - 0.62)
    above_head_reward = 1.0 / (1.0 + jp.exp(-above_x))

    # 6) Elbow up encouragement: target elbow_z > 0.55 when hand is high
    elbow_target = 0.55
    elbow_err = elbow_z_est - elbow_target
    elbow_reward = jp.exp(-elbow_err**2 / (0.1**2)) * height_gate

    return {
        "hand_height": height_reward,
        "hand_rise": rise_reward,
        "hand_wave_y": wave_reward_gated,
        "hand_above_head": above_head_reward,
        "elbow_up": elbow_reward,
    }
